\subsection*{Executive Summary}
The \texttt{deepMerge} function in \texttt{src/config.ts:11-27} creates a shallow copy of the \texttt{base} object (\texttt{const result = \{ ...base \}}) but then recursively merges nested objects \textbf{in place} — mutating the \texttt{result} object's nested properties rather than creating fresh copies at every level. When a nested object in \texttt{override} is merged into a nested object in \texttt{base}, the \texttt{result[key] = deepMerge(result[key], override[key])} assignment replaces the shallow copy with a merged object. However, if \texttt{result[key]} was a reference to a sub-object that was shared (e.g., loaded from a shared module cache), the mutation occurs on the shared reference.

Specifically:
1. \texttt{const result = \{ ...base \}} creates new top-level properties but nested objects are \textbf{shared references}
2. \texttt{deepMerge(result[key], override[key])} recursively \textbf{mutates} \texttt{result[key]} in place
3. If \texttt{result[key]} has properties that are also objects, \texttt{deepMerge} assigns them directly without copying

The impact is that the original \texttt{base} object's nested structures may be mutated when \texttt{override} provides nested values — a violation of the expected pure-functional merge semantics.

\subsection*{Root Cause Analysis}
\#\#\# The Code



\begin{lstlisting}[language=TypeScript]
// config.ts:11-27
function deepMerge(base: Record<string, any>, override: Record<string, any>): Record<string, any> {
    const result = { ...base };                           // ← Shallow copy of top level only
    for (const key of Object.keys(override)) {
        if (
            result[key] &&
            typeof result[key] === "object" &&
            !Array.isArray(result[key]) &&
            typeof override[key] === "object" &&
            !Array.isArray(override[key])
        ) {
            result[key] = deepMerge(result[key], override[key]);  // ← Recursive merge
        } else {
            result[key] = override[key];                  // ← Override value
        }
    }
    return result;
}

\end{lstlisting}

\#\#\# Real-World Scenario

In \texttt{loadEnvConfig}, the \texttt{base} object comes from \texttt{loadYamlFile} which reads from disk. If \texttt{base} is used elsewhere in the application (e.g., cached, referenced by another module), that shared reference will see mutated values after \texttt{deepMerge} is called. This can cause:

1. \textbf{Stale config leaks}: A cached base config object is mutated by a subsequent merge
2. \textbf{Race conditions}: If two merges happen concurrently on the same base
3. \textbf{Hard-to-debug state}: The original \texttt{default.yaml} values are silently overwritten

\subsection*{Fix Implementation}
\#\#\# Option A: Deep Clone at Every Level (Recommended)



\begin{lstlisting}[language=TypeScript]
// config.ts — deep clone to prevent mutation
function deepClone<T>(obj: T): T {
    if (obj === null || typeof obj !== "object") return obj;
    if (Array.isArray(obj)) return obj.map(deepClone) as any;
    const cloned: Record<string, any> = {};
    for (const key of Object.keys(obj as Record<string, any>)) {
        cloned[key] = deepClone((obj as Record<string, any>)[key]);
    }
    return cloned as T;
}

function deepMerge(base: Record<string, any>, override: Record<string, any>): Record<string, any> {
    const result = deepClone(base);                    // ← Deep clone base first
    for (const key of Object.keys(override)) {
        if (
            result[key] &&
            typeof result[key] === "object" &&
            !Array.isArray(result[key]) &&
            typeof override[key] === "object" &&
            !Array.isArray(override[key])
        ) {
            result[key] = deepMerge(result[key], override[key]);
        } else {
            result[key] = deepClone(override[key]);    // ← Clone override values too
        }
    }
    return result;
}

\#\#\# Option C: Pure Recursive Merge Without Mutation



\begin{lstlisting}[language=TypeScript]
// config.ts — purely functional merge
function deepMerge(base: Record<string, any>, override: Record<string, any>): Record<string, any> {
    const merged: Record<string, any> = {};

    // Copy all keys from base
    for (const key of Object.keys(base)) {
        merged[key] = base[key];
    }

    // Merge or override from override
    for (const key of Object.keys(override)) {
        if (key in merged) {
            if (
                typeof merged[key] === "object" &&
                !Array.isArray(merged[key]) &&
                typeof override[key] === "object" &&
                !Array.isArray(override[key]) &&
                merged[key] !== null &&
                override[key] !== null
            ) {
                merged[key] = deepMerge(merged[key], override[key]);
            } else {
                merged[key] = override[key];
            }
        } else {
            merged[key] = override[key];
        }
    }

    return merged;
}

\subsection*{Verification}

// verification/bug-035-verify.ts
import { deepMerge } from "../src/config.js";

// If deepMerge is not exported, test the logic directly
function deepClone<T>(obj: T): T {
    if (obj === null || typeof obj !== "object") return obj;
    if (Array.isArray(obj)) return obj.map(deepClone) as unknown as T;
    const cloned: Record<string, any> = {};
    for (const key of Object.keys(obj as Record<string, any>)) {
        cloned[key] = deepClone((obj as Record<string, any>)[key]);
    }
    return cloned as T;
}

function deepMergeFixed(base: Record<string, any>, override: Record<string, any>): Record<string, any> {
    const result = deepClone(base);
    for (const key of Object.keys(override)) {
        if (
            result[key] !== undefined &&
            result[key] !== null &&
            typeof result[key] === "object" &&
            !Array.isArray(result[key]) &&
            typeof override[key] === "object" &&
            !Array.isArray(override[key]) &&
            override[key] !== null
        ) {
            result[key] = deepMergeFixed(result[key], override[key]);
        } else {
            result[key] = deepClone(override[key]);
        }
    }
    return result;
}

function verify() {
    let failures = 0;

    // Test 1: Base is not mutated
    console.log("Test 1: Base is not mutated after merge...");
    const base1 = {
        model: {
            preferred: "gpt-4",
            settings: { temperature: 0.7 }
        }
    };
    const override1 = {
        model: {
            preferred: "claude-3",
            settings: { temperature: 0.5 }
        }
    };
    const originalTemp = base1.model.settings.temperature;
    const merged1 = deepMergeFixed(base1, override1);

    if (base1.model.settings.temperature === originalTemp && merged1.model.settings.temperature === 0.5) {
        console.log("  PASS: Base temperature unchanged, merged temperature updated");
    } else {
        console.log(`  FAIL: Base=${base1.model.settings.temperature}, Merged=${merged1.model.settings.temperature}`);
        failures++;
    }

    // Test 2: Override is not mutated
    console.log("Test 2: Override is not mutated...");
    const base2 = { a: { b: 1, c: 2 } };
    const override2 = { a: { d: 3 } };
    const merged2 = deepMergeFixed(base2, override2);

    if (!("d" in base2.a)) {
        console.log("  PASS: Override properties not added to base");
    } else {
        console.log("  FAIL: base.a has d");
        failures++;
    }

    // Test 3: Nested object merge works correctly
    console.log("Test 3: Nested merge produces correct result...");
    const base3 = {
        log_level: "info",
        model: {
            preferred: "gpt-4",
            fallback: ["gpt-3.5"],
            settings: { temperature: 0.7, max_tokens: 2000 }
        },
        tools: ["read", "write"],
    };
    const override3 = {
        log_level: "debug",
        model: {
            preferred: "claude-3",
            settings: { temperature: 0.3, top_p: 0.9 }
        }
    };
    const merged3 = deepMergeFixed(base3, override3);

    const expected3 = {
        log_level: "debug",
        model: {
            preferred: "claude-3",
            fallback: ["gpt-3.5"],
            settings: { temperature: 0.3, max_tokens: 2000, top_p: 0.9 }
        },
        tools: ["read", "write"],
    };

    if (JSON.stringify(merged3) === JSON.stringify(expected3)) {
        console.log("  PASS: Merged result matches expected");
    } else {
        console.log("  FAIL: Merged result differs");
        console.log(`  Got:      ${JSON.stringify(merged3)}`);
        console.log(`  Expected: ${JSON.stringify(expected3)}`);
        failures++;
    }

    // Test 4: Base remains identical to original
    console.log("Test 4: Base object identity preserved...");
    const base4 = { x: { y: { z: 1 } } };
    const override4 = { a: { b: 2 } };
    const originalBaseStr = JSON.stringify(base4);
    deepMergeFixed(base4, override4);
    const afterMergeStr = JSON.stringify(base4);

    if (originalBaseStr === afterMergeStr) {
        console.log("  PASS: Base object unchanged after merge");
    } else {
        console.log("  FAIL: Base was modified");
        failures++;
    }

    // Test 5: Null values don't crash
    console.log("Test 5: Null values handled gracefully...");
    const base5 = { a: null, b: { c: 1 } };
    const override5 = { a: { d: 2 }, b: null };
    try {
        const merged5 = deepMergeFixed(base5, override5);
        if (merged5.a !== null && merged5.a.d === 2 && merged5.b === null) {
            console.log("  PASS: Null values handled correctly");
        } else {
            console.log("  FAIL: Unexpected result");
            failures++;
        }
    } catch (e: any) {
        console.log(`  FAIL: Exception: ${e.message}`);
        failures++;
    }

    console.log(`\n${failures === 0 ? "ALL TESTS PASSED" : `${failures} TEST(S) FAILED`}`);
    process.exit(failures > 0 ? 1 : 0);
}

verify();
